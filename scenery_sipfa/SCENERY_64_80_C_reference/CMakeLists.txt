project(scenery_reference_c VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)

set(CMAKE_C_EXTENSIONS OFF)

add_library(scenery STATIC src/scenery.c src/persistent_fault.c src/detection_dataset.c src/known_detection_attack.c src/unknown_detection_attack.c src/infection_dataset.c src/known_infection_attack.c src/unknown_infection_attack.c)
target_include_directories(scenery PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
    target_compile_options(scenery PRIVATE /W4)
else()
    target_compile_options(scenery PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
endif()

    target_link_libraries(${test_name} PRIVATE scenery)
endforeach()
